<body>

  <h2>Price Lookup</h2>
  
  <div class="search-container">
    <input type="text" id="searchBar" placeholder="🔍 Search item name..." onkeyup="searchItems()">
  </div>

  <div id="resultsList"></div>

  <!-- This is the hidden owner section -->
  <div class="admin-panel" id="adminPanel">
    <h3>Owner: Update Price</h3>
    <input type="text" id="itemName" placeholder="Item Name (e.g., Basin Tap)">
    <input type="number" id="itemPrice" placeholder="Price (₹)">
    <button id="saveBtn" onclick="saveItem()">Save Price</button>
  </div>

  <!-- The secret button to unlock the owner section -->
  <div class="secret-lock" onclick="checkPin()">🔒</div>

  <script>
    // Your specific Google Web App URL is correctly inserted here
    const API_URL = "https://script.google.com/macros/s/AKfycbwvQzcRgg10_EWOaJD49VQisYhpH9fG5jhqdAN2hPPZPFMDBMV1u9NH12W1kjo9PQM0/exec";
    
    // Set your secret PIN here
    const ADMIN_PIN = "1234"; 
    
    let inventory = {};

    async function loadItems() {
      const results = document.getElementById('resultsList');
      results.innerHTML = '<div style="text-align:center; color:#777; padding:20px;">Loading prices...</div>';
      
      try {
        const response = await fetch(API_URL);
        inventory = await response.json();
        displayItems();
      } catch (error) {
        results.innerHTML = '<div style="text-align:center; color:red; padding:20px;">Error loading. Check internet.</div>';
      }
    }

    function displayItems(filter = '') {
      const results = document.getElementById('resultsList');
      results.innerHTML = '';
      
      for (let item in inventory) {
        if (item.toLowerCase().includes(filter.toLowerCase())) {
          results.innerHTML += `
            <div class="item">
              <span>${item}</span>
              <strong>₹${inventory[item]}</strong>
            </div>
          `;
        }
      }
      
      if (results.innerHTML === '') {
        results.innerHTML = '<div style="text-align:center; color:#777; padding:20px;">No items found.</div>';
      }
    }

    function searchItems() {
      const query = document.getElementById('searchBar').value;
      displayItems(query);
    }

    function checkPin() {
      const userPin = prompt("Enter Owner PIN:");
      if (userPin === ADMIN_PIN) {
        document.getElementById('adminPanel').style.display = 'block';
      } else if (userPin !== null) {
        alert("Incorrect PIN.");
      }
    }

    async function saveItem() {
      const name = document.getElementById('itemName').value.trim();
      const price = document.getElementById('itemPrice').value;
      const btn = document.getElementById('saveBtn');
      
      if (name && price) {
        btn.innerText = "Saving...";
        btn.style.background = "#555";

        try {
          await fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify({ itemName: name, itemPrice: price }),
            headers: { 'Content-Type': 'text/plain;charset=utf-8' } 
          });
          
          inventory[name] = price;
          document.getElementById('itemName').value = '';
          document.getElementById('itemPrice').value = '';
          displayItems(document.getElementById('searchBar').value);
          
        } catch (error) {
          alert("Failed to save.");
        }
        
        btn.innerText = "Save Price";
        btn.style.background = "#007aff";
      } else {
        alert("Please enter both a name and a price.");
      }
    }

    loadItems();
  </script>
